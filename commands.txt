## script to visualize point cloud (3D) ##

python - <<'PY'
from pathlib import Path
import open3d as o3d

path = Path(
    "/home/cidis/Documents/GitHub/Depth-Anything-3/"
    "outputs/DA3/real/seq008_da3stream_p1920_n0/"
    "pcd/combined_pcd.ply"
)

if not path.exists():
    raise FileNotFoundError(path)

point_cloud = o3d.io.read_point_cloud(str(path))

print("Archivo:", path)
print("Número de puntos:", len(point_cloud.points))
print("Tiene colores:", point_cloud.has_colors())

if len(point_cloud.points) == 0:
    raise RuntimeError("La nube de puntos está vacía.")

o3d.visualization.draw_geometries(
    [point_cloud],
    window_name="DA3 — seq007",
    width=1280,
    height=720,
)
PY
